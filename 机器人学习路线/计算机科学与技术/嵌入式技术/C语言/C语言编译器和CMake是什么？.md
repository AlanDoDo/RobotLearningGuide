```c
#include <stdio.h>  // 头文件：引入标准输入输出库---就像准备PPT课件，没课件没法上课

// int main() 程序入口，就相当于上课铃响了

// 大括号{}里的代码，就像上课过程

int main() {        

    // 输出一句话 Hello
    // \n相当于讲完一个知识点，翻页了

    printf("Hello\n");    

    return 0;                   // 返回0：表示成功结束，交给操作系统

}
```

  
  

#include <stdio.h>  // 标准输入输出库：就像老师上课必备的PPT

  

// 自定义函数：提前声明（像课前预告明天要讲的内容）

// void 我要说

void printWelcome();     // 欢迎 函数

void printSuggestion(int totalMinutes);  // 根据总时间给出回答

  

int main() {  // 主函数：上课铃响了！程序从这里开始执行

    int lessonCount;                 // 变量1：今天有几门课？

    int lessonTime;                  // 变量2：每门课的时间（分钟）

    int total = 0;                   // 累加器：总上课时间，初始为0

    int i;                           // 循环计数器

    float average;                   // 平均每门课时间（用float支持小数）

    char dayType;                    // 周几？用switch判断（'M'周一，'F'周五等）

  

    printWelcome();  // 调用函数：像老师先问好

  

    printf("今天是周几？(M=周一, T=周二, W=周三, R=周四, F=周五): ");

    scanf(" %c", &dayType);  // 注意空格跳过前面的回车

  

    printf("今天有几门课？: ");

    scanf("%d", &lessonCount);

  

    // for循环：已知次数的重复，就像一节接一节地上课

    for(i = 1; i <= lessonCount; i++) {

        printf("第 %d 门课上了多少分钟？: ", i);  // 格式化输出，带编号

        scanf("%d", &lessonTime);

        // 输入验证：简单判断时间是否合理

        if(lessonTime < 0 || lessonTime > 200) {

            printf("    时间不合理！已忽略这门课。\n");

            continue;  // 跳过本次循环，像这节课被取消

        }

        total += lessonTime;  // 累加总时间

    }

  

    // 计算平均时间

    if(lessonCount > 0) {

        average = (float)total / lessonCount;  // 类型转换：int转float，避免整数除法

    } else {

        average = 0;

    }

  

    // 输出总结报告：像老师下课前总结

    printf("\n========== 上课总结报告 ==========\n");

    printf("总上课时间：%d 分钟（约 %.2f 小时）\n", total, total / 60.0);

    printf("平均每门课：%.2f 分钟\n", average);

    // 判断是否超负荷

    if(total > 300) {

        printf("警告：今天上课太多了！注意休息！\n");

    } else if(total > 180) {

        printf("中等强度，好好坚持！\n");

    } else {

        printf("轻松的一天，棒！\n");

    }

  

    // switch语句：根据星期给出不同建议

    printf("\n特别提醒：\n");

    switch(dayType) {

        case 'M':

        case 'm': printf("周一，别给自己太大压力，慢慢来~\n"); break;

        case 'F':

        case 'f': printf("周五！坚持住，周末就解放了！\n"); break;

        case 'W':

        case 'w': printf("周三，星期过半，继续加油！\n"); break;

        default:  printf("好好学习，天天向上！\n");

    }

  

    printSuggestion(total);  // 调用建议函数

  

    return 0;  // 下课铃响，程序正常结束

}

  

// 函数定义：真正实现功能的地方（像老师备课写好的讲义）

void printWelcome() {

    printf("====================================\n");

    printf("       欢迎使用时间管理器\n");

    printf("     帮你统计今天学习的时间~\n");

    printf("====================================\n");

}

  

void printSuggestion(int totalMinutes) {

    if(totalMinutes >= 360) {

        printf("你今天学了 %d 分钟，学霸！真厉害！\n", totalMinutes);

    } else {

        printf("今天学了 %d 分钟，继续加油，你可以的！\n", totalMinutes);

    }

}
